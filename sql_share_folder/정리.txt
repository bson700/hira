--DROP TABLE SALES2 PURGE  ;
CREATE TABLE SALES2 
AS 
SELECT S.*
  FROM SALES S
  CROSS JOIN
  (SELECT ROWNUM 
     FROM DUAL 
	CONNECT BY LEVEL <= 6) ;

CREATE INDEX SALE2_IX ON SALES2(CUST_ID) ;

EXECUTE DBMS_STATS.GATHER_TABLE_STATS(USER,'SALES2') ;
