SELECT A.BRANCH, A.AVG_AMT, COUNT(*) AS CNT 
  FROM (SELECT BRANCH, TRUNC(AVG(LN_AMT),-6) AS AVG_AMT 
          FROM TACCT 
         WHERE LMT_TYP IS NULL 
         GROUP BY BRANCH ) A 
  JOIN (SELECT BRANCH, TRUNC(AVG(LN_AMT),-6) AS AVG_AMT 
          FROM TACCT 
         WHERE LMT_TYP IS NULL 
         GROUP BY BRANCH ) B 
    ON A.AVG_AMT BETWEEN B.AVG_AMT - 5000000 AND B.AVG_AMT + 5000000 
GROUP BY A.BRANCH, A.AVG_AMT     
ORDER BY 2 ;

답안2
WITH TAVG AS (SELECT BRANCH, TRUNC(AVG(LN_AMT),-6) AS AVG_AMT 
                FROM TACCT 
               WHERE LMT_TYP IS NULL 
               GROUP BY BRANCH ) 
SELECT A.BRANCH, A.AVG_AMT, COUNT(*) AS CNT 
  FROM TAVG  A 
  JOIN TAVG  B 
    ON A.AVG_AMT BETWEEN B.AVG_AMT - 5000000 AND B.AVG_AMT + 5000000 
GROUP BY A.BRANCH, A.AVG_AMT     
ORDER BY 2 ;

SELECT BRANCH
      ,TRUNC(AVG(LN_AMT),-6) AS AVG_AMT
      ,COUNT(*) OVER(ORDER BY TRUNC(AVG(LN_AMT),-6)
                     RANGE BETWEEN 5000000 PRECEDING 
                               AND 5000000 FOLLOWING) AS CNT
  FROM TACCT 
 WHERE LMT_TYP IS NULL
 GROUP BY BRANCH 
 ORDER BY AVG_AMT ; 


SELECT BRANCH
      ,AVG_AMT
      ,COUNT(*) OVER(ORDER BY AVG_AMT
                     RANGE BETWEEN 5000000 PRECEDING 
                               AND 5000000 FOLLOWING) AS CNT
  FROM (SELECT BRANCH
      ,TRUNC(AVG(LN_AMT),-6) AS AVG_AMT
  FROM TACCT 
 WHERE LMT_TYP IS NULL
 GROUP BY BRANCH )
 ORDER BY AVG_AMT ;

-- 5일 이평선
SELECT 단축코드, 일자, 종가
      ,TRUNC(AVG(종가) OVER(ORDER BY 일자 DESC 
                            ROWS BETWEEN CURRENT ROW AND 4 FOLLOWING)) AS "이평선(5일)"
      ,TRUNC(AVG(종가) OVER(ORDER BY 일자 DESC 
                            RANGE BETWEEN CURRENT ROW AND 4 FOLLOWING)) AS "이평선(5일)"
  FROM 종목현황 ;
