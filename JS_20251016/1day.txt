    <script>
        // key, value를 전달받아 객체를 반환하는 함수
        function generateObject(key, value) {
            /*
            const obj = {};
            obj[key] = value;
            return obj;
            */

            return {[key]: value};
        }

        console.log(generateObject("name", "홍길동"));   // {name: "홍길동"}
    </script>    


    <script>
        // key, value, num를 전달받아 { key+num: value } 형태의객체를 반환하는 함수
        function generateObject(key, value, num) {
            /*
            const obj = {};
            obj[key+num] = value;
            return obj;
            */
            return {[key+num]: value};
        }

        console.log(generateObject("name", "홍길동", 1));   // {name1: "홍길동"}
        console.log(generateObject("name", "고길동", 2));   // {name2: "고길동"}
    </script>

객체 데이터를 순차적으로 처리하는 방버
    <script>
        const person = {
            name: "홍길동", 
            age: 23,
            email: "test@test.com",
        };
        console.log(person);

        // for-in => 객체 속성의 키를 반환
        for (let key in person) {
            console.log(`person["${key}"] => ${person[key]}`);
        }

        const arr = [ "첫번째", "두번째", "세번째" ];
        arr.name = "배열";
        for (let idx in arr) {
            console.log(`arr[${idx}] => ${arr[idx]}`);
        }

        // for-of ==> 배열 데이터(값)를 순차적으로 반환
        let idx = 0;
        for (let value of arr) {
            console.log(`arr[${idx}] => ${value}`);
            idx ++;
        }
    </script>

함수를 정의하는 방법
    <script>
        {
            // 함수 선언문 형태로 함수를 정의 => function 함수이름 (매개변수) { 함수본문 }
            function add (a, b) {
                return a + b;
            }

            // 함수 호출 => 함수이름(인자)
            console.log(add(10, 20));   // 30
        }

        {
            // 익명 함수 표현식으로 함수를 정의 => const 함수변수 = function (매개변수) { 함수본문 };
            const add = function (a, b) { return a + b; };

            // 함수 호출 => 함수변수(인자)
            console.log(add(20, 30));   // 50
        }

        {
            // 기명 함수 표현식으로 함수를 정의 => const 함수변수 = function 함수이름 (매개변수) { 함수본문 };
            const add = function sum (a, b) { return a + b; };

            // 함수 호출 => 함수변수(인자)
            console.log(add(30, 40));   // 70

            // 함수이름(인자)
            try {
                console.log(sum(40, 50));   
            } catch (e) {
                console.error(e);       // sum is not defined
            }
        }

        {
            // 기명 함수 표현식에서 함수 이름은 외부에서 직접 호출(사용)할 수 없으며, 
            // 함수 내부에서만 사용(호출 = 재귀호출)할 수 있음 
            const myfactorial = function factorial (n) {
                if (n <= 1) return 1;
                return n * factorial(n-1);
            };
            console.log(myfactorial(10));   // 3628800
        }

        {
            // Function 생성자 함수를 이용해서 함수를 정의 
            const sum = new Function("a", "b", "return a + b;");

            // 함수 실행
            console.log(sum(10, 20));
        }
    </script>
